// ============================================================ // D'Chica COGS Monitor — Google Apps Script v6 // v6 change: cost lookup uses FULL SKU including size suffix // (e.g. DCWNCPBJU8932/XL) so each size gets its own cost. // Freebie check still uses master SKU prefix (DCFR). // ============================================================ const COSTING_SHEET_ID = "1np-x-FaSeuK1omriLNnwpOJE6T163pPDIudGKUGrl9U"; const COUPON_SHEET_ID = "1t2G4TE1svhXbtio6jwmqa9obkUCchxZ3rH95kdQKHDA"; const FREEBIE_PREFIX = "DCFR"; const NEW_COST_COL = 8; // column I: "New Cost (Viscose)" // ── MENU ───────────────────────────────────────────────────── function onOpen() { SpreadsheetApp.getUi() .createMenu("📊 D'Chica COGS") .addItem("▶ Run COGS Report", "runCOGSReport") .addToUi(); } // ── MAIN ───────────────────────────────────────────────────── function runCOGSReport() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const ui = SpreadsheetApp.getUi(); const result = ui.prompt( "Paste the Google Drive File ID of the Shopify export (.csv)", "Open the file in Drive → copy the ID from the URL", ui.ButtonSet.OK_CANCEL ); if (result.getSelectedButton() !== ui.Button.OK) return; const fileId = result.getResponseText().trim(); if (!fileId) { ui.alert("No file ID entered."); return; } ss.toast("Reading export file…", "D'Chica COGS", -1); try { const file = DriveApp.getFileById(fileId); const name = file.getName().toLowerCase(); let rows, headers; if (name.endsWith(".csv")) { const text = file.getBlob().getDataAsString("UTF-8"); const parsed = Utilities.parseCsv(text); const result = parseRows(parsed); rows = result.rows; headers = result.headers; } else { ui.alert( "Please upload the raw Shopify export as .csv\n\n" + "Shopify → Orders → Export → choose CSV (do NOT open/resave in Excel first)." ); return; } if (!rows || rows.length === 0) { ui.alert("❌ File appears empty, or columns 'Name' and 'Lineitem sku' not found."); return; } const reportDate = detectReportDate(rows, headers); ss.toast("Loading cost sheet…", "D'Chica COGS", -1); const costMap = loadCostMap(); ss.toast("Calculating COGS for " + reportDate + "…", "D'Chica COGS", -1); const noFreebiecoupons = loadNoFreebiecoupons(reportDate); const summary = calculateCOGS(rows, headers, costMap, noFreebiecoupons); summary.date = reportDate; writeToSheet(ss, summary); ss.toast("✅ Done! See COGS Report tab.", "D'Chica COGS", 6); } catch(e) { ui.alert("❌ Error: " + e.message); throw e; } } // ── PARSING ─────────────────────────────────────────────────── function parseRows(rows) { if (rows.length < 2) return { rows: [], headers: {} }; let headers = {}, headerIdx = 0; for (let i = 0; i < Math.min(5, rows.length); i++) { const idx = {}; rows[i].forEach((h, j) => { if (h) idx[String(h).trim()] = j; }); if (idx["Name"] !== undefined && idx["Lineitem sku"] !== undefined) { headers = idx; headerIdx = i; break; } } return { rows: rows.slice(headerIdx + 1).filter(r => r.some(c => c !== "")), headers: headers }; } function detectReportDate(rows, headers) { // Prefer "Created at" since "Paid at" can be blank for pending orders const col = headers["Created at"] !== undefined ? headers["Created at"] : headers["Paid at"]; if (col === undefined) return Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd"); for (const row of rows) { if (!row[col]) continue; // Shopify format is often dd-mm-yyyy const raw = String(row[col]).trim(); let d = new Date(raw); if (isNaN(d.getTime())) { const m = raw.match(/^(\d{1,2})-(\d{1,2})-(\d{4})/); if (m) d = new Date(`${m[3]}-${m[2]}-${m[1]}`); } if (!isNaN(d.getTime())) return Utilities.formatDate(d, "Asia/Kolkata", "yyyy-MM-dd"); } return Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd"); } // ── COST SHEET LOOKUP ───────────────────────────────────────── function loadCostMap() { const sheet = SpreadsheetApp.openById(COSTING_SHEET_ID).getSheets()[0]; const data = sheet.getDataRange().getValues(); const map = {}; for (let i = 1; i < data.length; i++) { const sku = String(data[i][0] || "").trim(); if (!sku) continue; const newCost = parseFloat(String(data[i][NEW_COST_COL] || "").replace(/,/g,"").trim()); const currCost = parseFloat(String(data[i][1] || "").replace(/,/g,"").trim()); const cost = !isNaN(newCost) ? newCost : (!isNaN(currCost) ? currCost : null); if (cost !== null) map[sku] = cost; // last row wins for duplicates } return map; } function stripSize(sku) { // Returns master SKU without size (e.g. DCWNCPBJU8932/XL → DCWNCPBJU8932) // Used ONLY for freebie prefix check, not for cost lookup return String(sku || "").split("/")[0].trim(); } // ── COUPON LIST ─────────────────────────────────────────────── function loadNoFreebiecoupons(dateStr) { try { const sheet = SpreadsheetApp.openById(COUPON_SHEET_ID).getSheets()[0]; const data = sheet.getDataRange().getValues(); const active = new Set(); const target = new Date(dateStr + "T12:00:00"); for (let i = 1; i < data.length; i++) { const code = String(data[i][0] || "").trim().toUpperCase(); if (!code) continue; const from = data[i][1] ? new Date(data[i][1]) : new Date("2000-01-01"); const to = data[i][2] ? new Date(data[i][2]) : new Date("2099-12-31"); from.setHours(0,0,0,0); to.setHours(23,59,59,999); if (target >= from && target <= to) active.add(code); } return active; } catch(e) { return new Set(["FLAT50","FREESHAPEWEAR"]); } } // ── COGS CALCULATION ────────────────────────────────────────── function calculateCOGS(rows, headers, costMap, noFreebiecoupons) { const cName = headers["Name"]; const cQty = headers["Lineitem quantity"]; const cSku = headers["Lineitem sku"]; const cTotal = headers["Total"]; const cDisc = headers["Discount Code"]; // Order → discount, order → revenue (first non-empty row wins) const orderDisc = {}, orderRev = {}; rows.forEach(row => { const o = String(row[cName] || "").trim(); if (!o) return; if (orderDisc[o] === undefined) { orderDisc[o] = String(row[cDisc] || "").trim().toUpperCase(); } if (orderRev[o] === undefined && cTotal !== undefined) { const r = parseFloat(String(row[cTotal] || "").replace(/[₹,]/g,"").trim()); if (!isNaN(r) && r > 0) orderRev[o] = r; } }); let totalCOGS = 0, totalRev = 0, totalQty = 0; let exclCOGS = 0, exclUnits = 0; const missing = new Set(), triggered = new Set(), seen = new Set(); rows.forEach(row => { const o = String(row[cName] || "").trim(); if (!o) return; // v6 FIX: use full SKU (with size) for cost lookup const fullSku = String(row[cSku] || "").trim(); const master = stripSize(fullSku); // used only for freebie prefix check const qty = parseInt(row[cQty] || 0) || 0; const cost = costMap[fullSku]; // ← full SKU lookup e.g. DCWNCPBJU8932/XL const disc = orderDisc[o] || ""; const excl = noFreebiecoupons.has(disc); const freebie = master.startsWith(FREEBIE_PREFIX); const skip = freebie && excl; if (cost === undefined) { if (fullSku) missing.add(fullSku); } else { const icogs = cost * qty; if (skip) { exclCOGS += icogs; exclUnits += qty; triggered.add(disc); } else { totalCOGS += icogs; totalQty += qty; } } if (!seen.has(o)) { seen.add(o); totalRev += orderRev[o] || 0; } }); return { orders: seen.size, qty: totalQty, revenue: Math.round(totalRev * 100) / 100, cogs: Math.round(totalCOGS * 100) / 100, cogsPct: totalRev ? totalCOGS / totalRev : 0, exclCOGS: Math.round(exclCOGS * 100) / 100, exclUnits: exclUnits, triggered: [...triggered].join(", ") || "none", missing: [...missing].join(", ") || "—", }; } // ── WRITE TO SHEET ──────────────────────────────────────────── function ensureHeaders(ss) { let sheet = ss.getSheetByName("COGS Report"); if (!sheet) sheet = ss.insertSheet("COGS Report"); const a1 = sheet.getRange(1,1).getValue(); if (a1 !== "Date") { const headers = [ "Date","Orders","Units Sold","Revenue (₹)","COGS (₹)", "Gross Margin %","COGS %","Freebie COGS Removed (₹)", "Freebie Units Removed","Active No-Freebie Coupons","Missing SKUs" ]; sheet.getRange(1,1,1,headers.length) .setValues([headers]) .setFontWeight("bold") .setBackground("#1a1a2e") .setFontColor("#ffffff"); sheet.setFrozenRows(1); } return sheet; } function writeToSheet(ss, s) { const sheet = ensureHeaders(ss); const data = sheet.getDataRange().getValues(); let row = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === s.date) { row = i + 1; break; } } if (row === -1) row = sheet.getLastRow() + 1; sheet.getRange(row, 1, 1, 11).setValues([[ s.date, s.orders, s.qty, s.revenue, s.cogs, ((1 - s.cogsPct)*100).toFixed(1) + "%", (s.cogsPct*100).toFixed(2) + "%", s.exclCOGS, s.exclUnits, s.triggered, s.missing ]]); sheet.getRange(row, 4, 1, 2).setNumberFormat("₹#,##0.00"); sheet.getRange(row, 8, 1, 1).setNumberFormat("₹#,##0.00"); sheet.getRange(row, 1, 1, 11).setBackground(row % 2 === 0 ? "#f5f5f5" : "#ffffff"); sheet.autoResizeColumns(1, 11); ss.setActiveSheet(sheet); }
